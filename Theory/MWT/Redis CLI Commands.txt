If we want to implement in React

✅ Recommended (Correct) Approach — React + Node + Redis
============================================
STEP 1 — Backend (Node + Redis)
Install dependencies (in a separate folder, e.g., server/)
=============================================
mkdir server
cd server
npm init -y
npm install express ioredis cors
=============================================
server/index.js (API for React to call)
=============================================
const express = require("express");
const cors = require("cors");
const { Redis } = require("ioredis");

const app = express();
const client = new Redis(); // connects to 127.0.0.1:6379

app.use(cors());

app.get("/user/:id", async (req, res) => {
  const id = req.params.id;
  const value = await client.get(`user:${id}`);
  res.send(value);
});

app.listen(5000, () => console.log("Server running on 5000"));


=============================================
Run backend:
=============================================
node index.js

=============================================
✅ STEP 2 — React (Vite) Frontend
=============================================
Create Vite project:
=============================================
npm create vite@latest redis-react-app -- --template react

cd redis-react-app

npm install axios

npm run dev
====================================
src/App.jsx (React version of your code)
=========================================
import { useEffect, useState } from "react";
import axios from "axios";

function App() {
  const [value, setValue] = useState("");

  useEffect(() => {
    axios.get("http://localhost:5000/user/3")
      .then(res => {
        console.log("Result ->", res.data);
        setValue(res.data);
      })
      .catch(err => console.error(err));
  }, []);

  return (
    <h2>user:3 = {value || "Loading..."}</h2>
  );
}

export default App;
